missing_info = df.isnull().sum()
missing_percentage = (missing_info / len(df)) * 100
missing_info_df = pd.DataFrame({'Missing Values': missing_info, 'Percentage': missing_percentage})
missing_info_df = missing_info_df[missing_info_df['Missing Values'] > 0].sort_values(by='Percentage', ascending=False)

if not missing_info_df.empty:
    print("Missing Values and Percentages:")
    print(missing_info_df.round(4))
    plt.figure(figsize=(12, 6))
    sns.heatmap(df.isnull(), cbar=False, cmap='viridis')
    plt.title('Missing Values Heatmap')
    plt.show()
else:
    print("No missing values found in the dataset.")

df['order_date'] = pd.to_datetime(df['Date'], errors='coerce')
# Extract time features
df['order_year'] = df['order_date'].dt.year
df['order_month'] = df['order_date'].dt.month
df['order_weekday'] = df['order_date'].dt.day_name()
sns.set(style='whitegrid')

# Orders per Year
plt.figure(figsize=(4,3))
sns.countplot(data=df, x='order_year', palette='mako')
plt.title('Orders per Year')
plt.xlabel('Year')
plt.ylabel('Number of Orders')
plt.tight_layout()
plt.show()

# Orders per Month
plt.figure(figsize=(4,3))
sns.countplot(data=df, x='order_month', palette='viridis')
plt.title('Orders per Month')
plt.xlabel('Month')
plt.ylabel('Number of Orders')
plt.tight_layout()
plt.show()

# Orders by Weekday
plt.figure(figsize=(4,3))
order_days = ['Monday','Tuesday','Wednesday','Thursday','Friday','Saturday','Sunday']
sns.countplot(data=df, x='order_weekday', order=order_days, palette='magma')
plt.title('Orders by Day of the Week')
plt.xlabel('Weekday')
plt.ylabel('Number of Orders')
plt.xticks(rotation=45)
plt.tight_layout()
plt.show()
